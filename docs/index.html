Quick Start

# Clone the repo
git clone https://github.com/TileTopia-HQ/viewtopia.git
cd viewtopia

# Install dependencies
npm install

# Start dev server (port 5174)
npm run dev

# Open http://localhost:5174
# Backends are optional — viewer adapts to what's available

# Optional: start TileTopia (3D Tiles)
cd ../tiletopia && cargo run -- serve --port 3000

# Optional: start GeoLang (AI agent)
cd ../geolang && python -m uvicorn src.app:app --port 8080
